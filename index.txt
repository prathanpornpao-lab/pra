    <script>
        // ข้อมูลเกม 5 ชุด ตามการออกแบบ
        const gameData = [
            { id: 1, p_num1: 1, p_den1: 2, op: '+', p_num2: 1, p_den2: 3, c_denom: 6, a_num: 5, a_den: 6, bg_symbol: 'star', color: 'red' },
            { id: 2, p_num1: 3, p_den1: 4, op: '-', p_num2: 1, p_den2: 2, c_denom: 4, a_num: 1, a_den: 4, bg_symbol: 'circle', color: 'blue' },
            { id: 3, p_num1: 2, p_den1: 3, op: '+', p_num2: 1, p_den2: 4, c_denom: 12, a_num: 11, a_den: 12, bg_symbol: 'triangle', color: 'green' },
            { id: 4, p_num1: 5, p_den1: 6, op: '-', p_num2: 1, p_den2: 3, c_denom: 6, a_num: 3, a_den: 6, bg_symbol: 'square', color: 'orange' },
            { id: 5, p_num1: 1, p_den1: 2, op: '+', p_num2: 2, p_den2: 5, c_denom: 10, a_num: 9, a_den: 10, bg_symbol: 'hexagon', color: 'purple' },
        ];

        // State ของเกม
        let currentRound = 0;
        let score = 0;
        let currentProblem = null;
        let availableCards = []; // การ์ดที่อยู่ในกองด้านล่าง
        let slotState = { denom: null, ans: null, img: null }; // การ์ดที่อยู่ในช่อง

        // ระบบเสียง (Tone.js)
        const synth = new Tone.PolySynth(Tone.Synth).toDestination();
        synth.volume.value = -10;

        function playSound(type) {
            if (Tone.context.state !== 'running') return;
            const now = Tone.now();
            switch(type) {
                case 'click':
                    synth.triggerAttackRelease("C5", "32n", now);
                    break;
                case 'pop':
                    synth.triggerAttackRelease("G4", "16n", now);
                    break;
                case 'success':
                    synth.triggerAttackRelease(["C4", "E4", "G4", "C5"], "8n", now);
                    break;
                case 'error':
                    synth.triggerAttackRelease("C3", "8n", now);
                    setTimeout(() => synth.triggerAttackRelease("G2", "4n", now + 0.1), 100);
                    break;
                case 'win':
                    synth.triggerAttackRelease(["C4", "E4", "G4"], "8n", now);
                    synth.triggerAttackRelease(["E4", "G4", "C5"], "8n", now + 0.2);
                    synth.triggerAttackRelease(["G4", "C5", "E5"], "4n", now + 0.4);
                    break;
            }
        }

        // ฟังก์ชันสร้าง HTML เศษส่วน
        function renderHTMLFraction(num, den, sizeClass = "text-xl") {
            return `
                <div class="fraction ${sizeClass}">
                    <span class="fraction-num">${num}</span>
                    <span class="fraction-den">${den}</span>
                </div>
            `;
        }

        // ฟังก์ชันสร้างภาพแถบเศษส่วน (SVG Fraction Bar)
        function generateFractionBarSVG(num, den) {
            // คำนวณความกว้างรวมให้สมดุล
            const boxWidth = 30;
            const height = 30;
            const totalWidth = den * boxWidth;
            // จำกัดความกว้างสูงสุดเพื่อให้ไม่ล้นการ์ด
            const viewBoxWidth = Math.max(totalWidth, 120); 
            const startX = (viewBoxWidth - totalWidth) / 2; // จัดกึ่งกลาง

            let rects = '';
            for(let i=0; i<den; i++) {
                const isFilled = i < num;
                const fillClass = isFilled ? "fill-pink-500" : "fill-white";
                rects += `<rect x="${startX + (i*boxWidth)}" y="10" width="${boxWidth}" height="${height}" rx="2" class="${fillClass} stroke-pink-700 stroke-2" />`;
            }
            
            return `
                <svg width="100%" height="50" viewBox="0 0 ${viewBoxWidth} 50" class="svg-fraction">
                    ${rects}
                </svg>
            `;
        }

        // สุ่ม Array
        function shuffleArray(array) {
            for (let i = array.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [array[i], array[j]] = [array[j], array[i]];
            }
            return array;
        }

        // เตรียมข้อมูลการ์ดสำหรับรอบนั้นๆ (ของจริง + ตัวหลอก)
        function prepareRoundCards(correctSet) {
            let cards = [];
            
            // 1. ใส่การ์ดที่ถูกต้อง
            cards.push({ id: `c_${correctSet.id}_denom`, matchId: correctSet.id, type: 'denom', value: correctSet.c_denom });
            cards.push({ id: `c_${correctSet.id}_ans`, matchId: correctSet.id, type: 'ans', num: correctSet.a_num, den: correctSet.a_den });
            cards.push({ id: `c_${correctSet.id}_img`, matchId: correctSet.id, type: 'img', num: correctSet.a_num, den: correctSet.a_den });

            // 2. สุ่มหาตัวหลอก 2 ชุด จากชุดอื่น
            const otherSets = shuffleArray(gameData.filter(d => d.id !== correctSet.id)).slice(0, 2);
            
            otherSets.forEach((wrongSet, idx) => {
                // หลอกส่วนร่วม
                cards.push({ id: `w_denom_${idx}`, matchId: wrongSet.id, type: 'denom', value: wrongSet.c_denom });
                // หลอกคำตอบ
                cards.push({ id: `w_ans_${idx}`, matchId: wrongSet.id, type: 'ans', num: wrongSet.a_num, den: wrongSet.a_den });
                // หลอกภาพ
                cards.push({ id: `w_img_${idx}`, matchId: wrongSet.id, type: 'img', num: wrongSet.a_num, den: wrongSet.a_den });
            });

            return shuffleArray(cards);
        }

        // แสดงโจทย์
        function renderProblem() {
            const container = document.getElementById('problem-container');
            const f1 = renderHTMLFraction(currentProblem.p_num1, currentProblem.p_den1);
            const f2 = renderHTMLFraction(currentProblem.p_num2, currentProblem.p_den2);
            const opColor = currentProblem.op === '+' ? 'text-blue-600' : 'text-red-500';
            container.innerHTML = `${f1} <span class="mx-2 ${opColor}">${currentProblem.op}</span> ${f2}`;
        }

        // สร้าง HTML ของการ์ด
        function createCardHTML(card) {
            let bgClass = '';
            let contentHTML = '';
            let iconHTML = '';

            if (card.type === 'denom') {
                bgClass = 'bg-yellow-100 border-yellow-400 hover:bg-yellow-200 text-yellow-800';
                contentHTML = `<div class="text-3xl font-bold">${card.value}</div>`;
                iconHTML = `<i class="fas fa-layer-group text-yellow-500 opacity-20 text-4xl absolute -bottom-2 -right-2"></i>`;
            } else if (card.type === 'ans') {
                bgClass = 'bg-green-100 border-green-400 hover:bg-green-200 text-green-800';
                contentHTML = renderHTMLFraction(card.num, card.den, "text-2xl");
                iconHTML = `<i class="fas fa-check-circle text-green-500 opacity-20 text-4xl absolute -bottom-2 -right-2"></i>`;
            } else if (card.type === 'img') {
                bgClass = 'bg-pink-100 border-pink-400 hover:bg-pink-200';
                contentHTML = generateFractionBarSVG(card.num, card.den);
                // iconHTML = `<i class="fas fa-image text-pink-500 opacity-20 text-4xl absolute -bottom-2 -right-2"></i>`;
            }

            return `
                <div class="card-pop relative flex flex-col items-center justify-center w-full h-24 md:h-28 rounded-xl border-b-4 border-2 ${bgClass} overflow-hidden" 
                     onclick="handleCardClick('${card.id}')">
                    ${contentHTML}
                    ${iconHTML}
                </div>
            `;
        }

        // เรนเดอร์การ์ดทั้งหมด
        function renderUI() {
            const choicesContainer = document.getElementById('choices-container');
            choicesContainer.innerHTML = '';
            
            // เรนเดอร์กองการ์ด
            availableCards.forEach(card => {
                const wrapper = document.createElement('div');
                wrapper.innerHTML = createCardHTML(card);
                choicesContainer.appendChild(wrapper.firstElementChild);
            });

            // เรนเดอร์ช่อง Slots
            ['denom', 'ans', 'img'].forEach(type => {
                const slot = document.getElementById(`slot-${type}`);
                const currentCard = slotState[type];
                
                if (currentCard) {
                    slot.innerHTML = createCardHTML(currentCard);
                    slot.classList.remove('slot-empty');
                    slot.classList.add('bg-white', 'shadow-inner');
                } else {
                    let typeName = type === 'denom' ? 'ส่วนร่วม' : type === 'ans' ? 'คำตอบ' : 'ภาพเศษส่วน';
                    slot.innerHTML = `<span class="text-slate-400 text-sm">แตะเลือกการ์ด${typeName}</span>`;
                    slot.classList.add('slot-empty');
                    slot.classList.remove('bg-white', 'shadow-inner');
                }
            });

            checkAllSlotsFilled();
        }

        // จัดการเมื่อแตะการ์ด
        function handleCardClick(cardId) {
            playSound('click');
            
            // หาว่าการ์ดนี้อยู่ใน Slot หรืออยู่ข้างล่าง
            let cardObj = null;
            let currentSlotType = null;

            // เช็คใน slots ก่อน
            for (const key in slotState) {
                if (slotState[key] && slotState[key].id === cardId) {
                    cardObj = slotState[key];
                    currentSlotType = key;
                    break;
                }
            }

            if (currentSlotType) {
                // ถ้าอยู่ใน slot ให้เอาออกไปคืนกอง
                slotState[currentSlotType] = null;
                availableCards.push(cardObj);
            } else {
                // ถ้าอยู่ในกอง ให้หา object และย้ายไป slot ที่ตรง type
                const index = availableCards.findIndex(c => c.id === cardId);
                if (index > -1) {
                    cardObj = availableCards[index];
                    const targetType = cardObj.type;
                    
                    // ถ้า slot นั้นมีการ์ดอยู่แล้ว ให้เอาการ์ดเดิมกลับมากองก่อน
                    if (slotState[targetType]) {
                        availableCards.push(slotState[targetType]);
                    }
                    
                    slotState[targetType] = cardObj;
                    availableCards.splice(index, 1);
                }
            }

            renderUI();
        }

        // ตรวจสอบว่าใส่การ์ดครบหรือยัง เพื่อโชว์ปุ่มตรวจคำตอบ
        function checkAllSlotsFilled() {
            const btnCheck = document.getElementById('btn-check');
            if (slotState.denom && slotState.ans && slotState.img) {
                btnCheck.classList.remove('hidden');
                btnCheck.classList.add('bounce-in');
            } else {
                btnCheck.classList.add('hidden');
                btnCheck.classList.remove('bounce-in');
            }
        }

        // ตรวจคำตอบ
        function checkAnswer() {
            const expectedId = currentProblem.id;
            const isDenomCorrect = slotState.denom.matchId === expectedId;
            const isAnsCorrect = slotState.ans.matchId === expectedId;
            const isImgCorrect = slotState.img.matchId === expectedId;

            if (isDenomCorrect && isAnsCorrect && isImgCorrect) {
                // ถูกต้องทั้งหมด
                playSound('success');
                score += 1;
                document.getElementById('score-display').textContent = score;
                
                showModal(
                    "🎉 เก่งมาก!", 
                    "สัญลักษณ์ด้านหลังการ์ดตรงกัน แสดงว่าจับคู่ถูกต้อง", 
                    "⭐ ทำข้อต่อไป", 
                    () => {
                        currentRound++;
                        if (currentRound < gameData.length) {
                            startRound(currentRound);
                        } else {
                            endGame();
                        }
                    },
                    "text-emerald-500"
                );
            } else {
                // มีจุดผิด
                playSound('error');
                
                // หาว่าผิดตรงไหนเพื่อใบ้
                let hint = "ลองตรวจสอบดูใหม่นะ<br>";
                if (!isDenomCorrect) hint += "- ตัวส่วนร่วมยังไม่ใช่นะ ลองหาเลขที่ทั้งสองส่วนหารลงตัวดู<br>";
                if (!isAnsCorrect && isDenomCorrect) hint += "- ส่วนร่วมถูกแล้ว แต่คำตอบยังคลาดเคลื่อนอยู่<br>";
                if (!isImgCorrect && isAnsCorrect) hint += "- คำตอบถูกแล้ว แต่เลือกภาพเศษส่วนไม่ตรงกับคำตอบนะ<br>";

                // ใส่เอฟเฟกต์สั่น
                document.querySelectorAll('[id^="slot-"]').forEach(el => {
                    el.classList.add('shake-animation');
                    setTimeout(() => el.classList.remove('shake-animation'), 500);
                });

                showModal("🤔 เกือบถูกแล้ว", hint, "ลองใหม่", null, "text-orange-500", "fas fa-exclamation-circle");
            }
        }

        // เริ่มรอบใหม่
        function startRound(roundIndex) {
            currentProblem = gameData[roundIndex];
            document.getElementById('round-display').textContent = roundIndex + 1;
            
            // ล้าง Slots
            slotState = { denom: null, ans: null, img: null };
            
            // เตรียมการ์ด
            availableCards = prepareRoundCards(currentProblem);
            
            renderProblem();
            renderUI();
        }

        // จบเกม
        function endGame() {
            playSound('win');
            let message = `คุณทำคะแนนได้ ${score} / ${gameData.length} คะแนน`;
            let icon = "🏆";
            if(score === gameData.length) message += "<br>ยอดเยี่ยม! เข้าใจเรื่องเศษส่วนได้อย่างทะลุปรุโปร่ง";
            else if (score >= 3) message += "<br>ทำได้ดีมาก! ฝึกอีกนิดจะเก่งขึ้นแน่นอน";
            else message += "<br>ไม่ต้องเสียใจ ลองเล่นใหม่อีกครั้งนะ";

            showModal("จบเกม!", message, "เล่นอีกครั้ง", () => {
                startGame();
            }, "text-yellow-500", null, true);
        }

        // ระบบ Modal แทน alert()
        const modalContainer = document.getElementById('custom-modal');
        const modalContent = document.getElementById('modal-content');
        const modalBtn = document.getElementById('modal-btn');
        let modalCallback = null;

        function showModal(title, desc, btnText, callback, colorClass = "text-blue-500", iconClass = null, isHtml = false) {
            document.getElementById('modal-title').textContent = title;
            document.getElementById('modal-title').className = `text-2xl font-bold mb-2 ${colorClass}`;
            
            if(isHtml) document.getElementById('modal-desc').innerHTML = desc;
            else document.getElementById('modal-desc').textContent = desc;
            
            document.getElementById('modal-btn').textContent = btnText;
            
            if (iconClass) {
                document.getElementById('modal-icon').innerHTML = `<i class="${iconClass} ${colorClass}"></i>`;
            } else {
                document.getElementById('modal-icon').innerHTML = title.includes("🎉") ? "🎉" : (title.includes("🏆") ? "🏆" : "💡");
            }

            modalCallback = callback;
            
            modalContainer.classList.remove('hidden');
            // Allow display block to apply before fading in
            setTimeout(() => {
                modalContainer.classList.remove('opacity-0');
                modalContent.classList.remove('scale-90');
            }, 10);
        }

        function closeModal() {
            playSound('pop');
            modalContainer.classList.add('opacity-0');
            modalContent.classList.add('scale-90');
            
            setTimeout(() => {
                modalContainer.classList.add('hidden');
                if (modalCallback) modalCallback();
            }, 300);
        }

        modalBtn.addEventListener('click', closeModal);

        // การเริ่มเกมครั้งแรก (ผูกกับปุ่ม Start เพื่อ init Tone.js)
        async function startGame() {
            // Start Audio Context
            await Tone.start();
            playSound('pop');
            
            // Hide Start Screen
            const startScreen = document.getElementById('start-screen');
            startScreen.classList.add('opacity-0');
            setTimeout(() => {
                startScreen.classList.add('hidden');
                document.getElementById('game-screen').classList.remove('hidden');
                
                // รีเซ็ตค่าและเริ่มรอบแรก
                currentRound = 0;
                score = 0;
                document.getElementById('score-display').textContent = score;
                startRound(currentRound);
            }, 500);
        }

        // Event Listeners
        document.getElementById('btn-start').addEventListener('click', startGame);
        document.getElementById('btn-check').addEventListener('click', checkAnswer);
